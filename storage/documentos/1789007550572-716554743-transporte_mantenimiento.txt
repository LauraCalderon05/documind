-- 1. Crear y seleccionar base de datos
CREATE DATABASE IF NOT EXISTS transporte_mantenimiento;
USE transporte_mantenimiento;

-- 2. Crear tablas
CREATE TABLE usuarios (
    cedula VARCHAR(20) PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    correo VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    rol ENUM('Administrador', 'Mecanico') NOT NULL,
    activo BOOLEAN DEFAULT TRUE
);

CREATE TABLE vehiculos (
    id_vehiculo INT AUTO_INCREMENT PRIMARY KEY,
    placa VARCHAR(20) UNIQUE NOT NULL,
    marca VARCHAR(50) NOT NULL,
    modelo VARCHAR(50) NOT NULL,
    anio INT NOT NULL,
    estado ENUM('Activo', 'Inactivo', 'En Mantenimiento') NOT NULL,
    activo BOOLEAN DEFAULT TRUE
);

CREATE TABLE mantenimientos (
    id_mantenimiento INT AUTO_INCREMENT PRIMARY KEY,
    id_vehiculo INT NOT NULL,
    cedula_usuario VARCHAR(20) NOT NULL,
    fecha DATE NOT NULL,
    fecha_fin DATE NULL,
    tipo VARCHAR(50) NOT NULL,
    descripcion TEXT,
    costo DECIMAL(10,2),
    estado_mantenimiento ENUM('En proceso', 'Finalizado') DEFAULT 'En proceso',
    FOREIGN KEY (id_vehiculo) REFERENCES vehiculos(id_vehiculo),
    FOREIGN KEY (cedula_usuario) REFERENCES usuarios(cedula)
);
